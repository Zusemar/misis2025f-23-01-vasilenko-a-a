cmake_minimum_required(VERSION 3.20)
project(walnut_segmentation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
add_library(walnut_core src/segmentation.cpp)
target_include_directories(walnut_core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(walnut_core PUBLIC ${OpenCV_LIBS})
target_compile_options(walnut_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(walnut_segment src/main.cpp)
target_link_libraries(walnut_segment PRIVATE walnut_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(walnut_tests tests/test_segmentation.cpp)
  target_link_libraries(walnut_tests PRIVATE walnut_core)
  add_test(NAME walnut_semantics COMMAND walnut_tests)
endif()
